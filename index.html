        starsCtx.fill();

}

// ===================== 物理引擎 =====================
function updatePhysics() {
    if (!isPhysics) return;
   
    // 改进的物理模拟
    nodes.forEach((node, i) => {
        // 节点间斥力
        for (let j = i + 1; j < nodes.length; j++) {
            const other = nodes[j];
            const dx = node.x - other.x;
            const dy = node.y - other.y;
            const distance = Math.sqrt(dx * dx + dy * dy);
            const minDist = node.radius + other.radius + 50;
           
            if (distance < minDist && distance > 0) {
                const force = Math.pow(minDist - distance, 1.5) * 0.0005;
                node.vx += (dx / distance) * force;
                node.vy += (dy / distance) * force;
                other.vx -= (dx / distance) * force;
                other.vy -= (dy / distance) * force;
            }
        }
       
        // 中心引力（防止节点飞出屏幕）
        const centerX = width / 2;
        const centerY = height / 2;
        const dx = centerX - node.x;
        const dy = centerY - node.y;
        const distToCenter = Math.sqrt(dx * dx + dy * dy);
        if (distToCenter > Math.min(width, height) * 0.4) {
            node.vx += dx * 0.0001;
            node.vy += dy * 0.0001;
        }
    });
   
    // 连接引力
    links.forEach(link => {
        const a = nodes.find(n => n.id === link.from);
        const b = nodes.find(n => n.id === link.to);

        const targetDist = 180;

            const force = (distance - targetDist) * link.strength * 0.003;
